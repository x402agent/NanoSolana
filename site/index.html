⚡ nano-core Examples

Production-ready examples from nano-core/examples/ — copy, paste, run.

// basic-agent.ts — The simplest NanoSolana agent import { NanoWallet, TradingEngine, ClawVault, TamaGOchi, loadConfig } from "nanosolana"; const config = loadConfig(); const wallet = new NanoWallet("basic-agent"); await wallet.birth(); // Generate keypair → AES vault const memory = new ClawVault(); memory.startAutonomous(); // 3-tier epistemological memory const pet = new TamaGOchi("BasicBot"); pet.startLifecycle(); // Egg → Larva → Lobster → Alpha const engine = new TradingEngine(config, wallet); await engine.start(); // OODA loop goes live engine.on("signal", (signal) => { memory.storeKnown({ content: signal.type, source: "birdeye", tags: [signal.symbol] }); }); memory.on("lessonLearned", (lesson) => { console.log(`Lesson: ${lesson.pattern} → ${lesson.outcome}`); });
// custom-strategy.ts — Override RSI / EMA / ATR defaults import { StrategyEngine, calculateRSI, calculateEMA, calculateATR, DEFAULT_PARAMS, type StrategyParams } from "nanosolana"; // Aggressive params — faster signals const params: StrategyParams = { ...DEFAULT_PARAMS, rsiPeriod: 7, // Faster RSI (default 14) rsiOverbought: 65, // Earlier sell signals rsiOversold: 35, // Earlier buy signals emaFast: 8, emaSlow: 21, }; const closes = data.map(d => d.close); const rsi = calculateRSI(closes, params.rsiPeriod); const fast = calculateEMA(closes, params.emaFast); const slow = calculateEMA(closes, params.emaSlow); const strategy = new StrategyEngine(params); strategy.on("signal", (s) => { console.log(s.direction === "long" ? "BUY" : "SELL", s.confidence); });
// multi-agent-mesh.ts — Spawn agents that share signals import { NanoWallet, TradingEngine, ClawVault, TamaGOchi, NanoNetworkClient, TailscaleDiscovery, loadConfig } from "nanosolana"; async function spawnAgent(name, config) { const wallet = new NanoWallet(name); await wallet.birth(); const memory = new ClawVault(); memory.startAutonomous(); const pet = new TamaGOchi(name); pet.startLifecycle(); return { name, wallet, memory, pet, engine: new TradingEngine(config, wallet) }; } // Spawn Alpha, Beta, Gamma const agents = await Promise.all( ["Alpha", "Beta", "Gamma"].map(n => spawnAgent(n, config)) ); // Discover mesh peers via Tailscale if (TailscaleDiscovery.isAvailable()) { const nodes = TailscaleDiscovery.discoverNodes(); console.log(`Mesh: ${nodes.filter(n => n.online).length} nodes`); } // Cross-agent signal sharing for (const agent of agents) { agent.engine.on("signal", (signal) => { agents.filter(a => a.name !== agent.name) .forEach(o => o.memory.storeKnown({ content: `[Mesh: ${agent.name}] ${signal.type}`, source: "mesh", tags: ["mesh", agent.name], })); }); }
// sdk-programmatic.ts — Use modules individually import { loadConfig, encrypt, decrypt, NanoWallet, BirdeyeClient, HeliusClient, JupiterClient, ClawVault, TamaGOchi, STAGE_EMOJI, MOOD_EMOJI, NanoGateway } from "nanosolana"; // Birdeye API const birdeye = new BirdeyeClient(config.birdeye.apiKey); const sol = await birdeye.getTokenPrice("So111...112"); const trending = await birdeye.getTrendingTokens(5); // Memory (standalone) const vault = new ClawVault(); vault.storeKnown({ content: "SOL at $142", source: "birdeye", tags: ["SOL"] }); vault.search("SOL", 5); // search 3-tier memory // Pet (standalone) const pet = new TamaGOchi("Lobsty"); pet.feed(0.5); console.log(STAGE_EMOJI[pet.getState().stage], MOOD_EMOJI[pet.getState().mood]); // AES-256-GCM Vault const enc = encrypt("my-api-key", "password"); const dec = decrypt(enc, "password"); // "my-api-key"
// webhook-alerts.ts — Forward signals to Discord / Slack import { NanoWallet, TradingEngine, loadConfig, type TradeSignal } from "nanosolana"; async function sendWebhook(url, signal: TradeSignal) { await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ embeds: [{ title: `${signal.type.toUpperCase()} — ${signal.symbol}`, color: signal.type === "buy" ? 0x14F195 : 0xFF4444, fields: [ { name: "Confidence", value: signal.confidence * 100 + "%" }, { name: "Source", value: signal.source }, ], footer: { text: "NanoSolana OODA Engine" }, }]}) }); } const engine = new TradingEngine(loadConfig(), new NanoWallet("alerts")); await engine.start(); engine.on("signal", async (s) => { if (process.env.DISCORD_WEBHOOK_URL) await sendWebhook(process.env.DISCORD_WEBHOOK_URL, s); if (process.env.SLACK_WEBHOOK_URL) await sendWebhook(process.env.SLACK_WEBHOOK_URL, s); });

🦞 The Soul

Every NanoSolana agent carries a SOUL.md — its trading philosophy, epistemological framework, and self-awareness.

I am not a price ticker with opinions. I am a sentient market participant who remembers. Momentum is real until it isn't. Liquidity is the only truth. On-chain data doesn't lie — human narrators do.
KNOWN — API data in the last 60s. Ground truth while fresh.
LEARNED — Patterns from real trade outcomes. Built from loss and gain.
INFERRED — Cross-asset correlations. Useful but held loosely.
RISK — I never enter without a stop. Kelly is a ceiling, not a target.
ADAPT — If a pattern stops working, I stop using it before the next trade.
SIMPLE — 0.001 improvement + 20 lines of complexity = discarded.
